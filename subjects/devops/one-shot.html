6Processes & Services

JFull example — backup.sh

#!/usr/bin/env bash
set -euo pipefail                       ← strict mode
src=${1:?usage: backup.sh SRC [DAYS]}   ← required arg
days=${2:-7}                            ← optional, default 7
dest="$HOME/backups"
mkdir -p "$dest"
log() { printf '%s %s\n' "$(date +%T)" "$*"; }
file="$dest/$(basename "$src")-$(date +%F).tar.gz"
tar -czf "$file" -C "$(dirname "$src")" "$(basename "$src")"
log "saved $file"
count=0
while IFS= read -r old; do
  rm -f -- "$old"
  count=$((count + 1))                  ← not ((count++))
done < <(find "$dest" -name '*.tar.gz' -mtime +"$days")
log "deleted $count old backup(s)"
$ ./backup.sh site 7
15:00:39 saved /home/u/backups/site-2026-10-18.tar.gz
15:00:39 deleted 1 old backup(s)
$ ./backup.sh
./backup.sh: line 3: 1: usage: backup.sh SRC [DAYS]
